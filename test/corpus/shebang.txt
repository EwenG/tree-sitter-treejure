==================
Basic Shebang
==================
#!/usr/bin/env bb
---
(source
  (shebang))

==================
Shebang with Code
==================
#!/usr/bin/clojure
(ns main)
---
(source
  (shebang)
  (list_literal
    (symbol (symbol_name))
    (symbol (symbol_name))))

==================

==================
 ; comment first

---

  (comment)

==================
Empty Shebang
==================
#!
(def x 1)
---
(source
  (shebang)
  (list_literal
    (symbol (symbol_name))
    (symbol (symbol_name))
    (number)))

==================
Shebang with Complex Gaps
==================
#!/usr/bin/bb

;; This is a script
;; Version 1.0

(println "Hello")
---
(source
  (shebang)
  (comment)
  (comment)
  (list_literal
    (symbol (symbol_name))
    (string)))
